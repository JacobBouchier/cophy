# How to combine the enabled tools into an ortholog call.

[neighborhood]
# Gene-neighborhood distance analysis: for each (species, primary, cooccur)
# triple, compute how many genes apart they are on the chromosome. Adds
# <id>_dist columns to the cooccurrence CSV (and per-primary distance
# columns to the collapsed CSV).
#
# Coordinate data comes from NCBI EFetch:
#   1. UniProt accession -> RefSeq protein id (from cached UniProt JSON, or
#      via NCBI esearch as fallback).
#   2. EFetch the RefSeq protein in GenBank format -> parse /coded_by= for
#      (contig, start, strand).
#   3. EFetch each unique contig's feature table -> sorted gene list to
#      convert coordinates into ordinals.
#
# All results are cached under cache_dir; subsequent runs reuse the cache
# and are fast. The FIRST run with this feature enabled can take an hour
# or more for large datasets (one NCBI call per accession + one per contig).
neighborhood          = false
# Distances are measured in NUCLEOTIDES between protein start coordinates.
# Beyond this cap, distances are rendered as '>N' (sign dropped). 100000bp
# (= 100kb) is generous and lets most operons / regulatory regions appear
# with exact numbers while still bounding the column width.
neighborhood_max_dist = 100000
